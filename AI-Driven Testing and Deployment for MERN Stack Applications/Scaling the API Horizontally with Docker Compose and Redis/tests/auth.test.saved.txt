import request from "supertest";
import bcrypt from "bcrypt";
import jwt from "jsonwebtoken";
import app from "../src/app.js";
import { env } from "../src/config/env.js";
import { User } from "../src/models/user.model.js";
import { connect, clear, close } from "./helpers/db.js";

beforeAll(async () => {
  await connect();
  await User.init();
}, 60_000);
afterEach(clear);
afterAll(close);

const ASHA = {
  name: "Asha",
  email: "asha@example.com",
  password: "CorrectHorse9!",
};

function register(fields = {}) {
  return request(app)
    .post("/api/v1/auth/register")
    .send({ ...ASHA, ...fields });
}

function login(fields = {}) {
  return request(app)
    .post("/api/v1/auth/login")
    .send({ email: ASHA.email, password: ASHA.password, ...fields });
}

describe("POST /api/v1/auth/register", () => {
  it("creates an account and returns 201 without the password", async () => {
    const res = await register({
      name: " Asha ",
      email: " Asha@Example.COM ",
    });

    expect(res.status).toBe(201);
    expect(res.body.message).toBe("Account created");
    expect(res.body.data).toMatchObject({
      name: "Asha",
      email: "asha@example.com",
    });
    expect(res.body.data._id).toEqual(expect.any(String));
    expect(res.body.data).not.toHaveProperty("password");
  });

  it("stores a salted bcrypt hash, never the plain password", async () => {
    await register();
    await register({ name: "Ben", email: "ben@example.com" });

    const users = await User.find().select("+password");
    const [asha, ben] = users.map((user) => user.password);

    expect(asha).toMatch(/^\$2b\$12\$.{53}$/);
    expect(asha).not.toBe(ASHA.password);
    expect(await bcrypt.compare(ASHA.password, asha)).toBe(true);
    expect(ben).not.toBe(asha);
  });

  it("returns 400 with a message for each invalid field", async () => {
    const res = await register({
      name: "",
      email: "not-an-email",
      password: "short",
    });

    expect(res.status).toBe(400);
    expect(res.body.errors).toEqual([
      { field: "name", message: "Name is required" },
      { field: "email", message: "Email must be a valid email address" },
      { field: "password", message: "Password must be at least 8 characters" },
    ]);
  });

  it("rejects a password longer than 72 bytes", async () => {
    const res = await register({ password: "é".repeat(40) });

    expect(res.status).toBe(400);
    expect(res.body.errors).toEqual([
      { field: "password", message: "Password must be at most 72 bytes" },
    ]);
  });

  it("returns 409 for an email that is already registered", async () => {
    await register();
    const res = await register({ name: "Asha Two", email: "ASHA@example.com" });

    expect(res.status).toBe(409);
    expect(res.body.message).toBe("A record with that email already exists");
    expect(await User.countDocuments()).toBe(1);
  });
});

describe("POST /api/v1/auth/login", () => {
  beforeEach(() => register());

  it("returns a signed token and the user without the password", async () => {
    const res = await login();

    expect(res.status).toBe(200);
    expect(res.body.message).toBe("Logged in");
    expect(res.body.data.user.email).toBe("asha@example.com");
    expect(res.body.data.user).not.toHaveProperty("password");

    const payload = jwt.verify(res.body.data.token, env.jwtSecret);
    expect(payload.sub).toBe(res.body.data.user._id);
    expect(Object.keys(payload).sort()).toEqual(["exp", "iat", "sub"]);
  });

  it("issues tokens that expire after the configured lifetime", async () => {
    const res = await login();
    const { iat, exp } = jwt.decode(res.body.data.token);

    expect(env.jwtExpiresIn).toBe("1h");
    expect(exp - iat).toBe(3600);
  });

  it("rejects a token signed with a different secret", async () => {
    const res = await login();

    expect(() => jwt.verify(res.body.data.token, "another-secret")).toThrow(
      "invalid signature",
    );
  });

  it("returns 401 for a wrong password", async () => {
    const res = await login({ password: "WrongPassword1!" });

    expect(res.status).toBe(401);
    expect(res.body).toEqual({
      status: "error",
      message: "Invalid email or password",
    });
  });

  it("returns the same 401 for an unknown email", async () => {
    const res = await login({ email: "nobody@example.com" });

    expect(res.status).toBe(401);
    expect(res.body).toEqual({
      status: "error",
      message: "Invalid email or password",
    });
  });

  it("returns 400 when email or password is missing", async () => {
    const res = await request(app).post("/api/v1/auth/login").send({});

    expect(res.status).toBe(400);
    expect(res.body.errors).toEqual([
      { field: "email", message: "Email is required" },
      { field: "password", message: "Password is required" },
    ]);
  });
});
